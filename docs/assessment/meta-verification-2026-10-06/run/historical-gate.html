# require.groups의 모든 묶음을 만족해야 통과.

rules:

  # ─── API surface 변경 → 계약 문서 + 릴리즈 공지 ─────────────────────────────
  - id: api-contract-sync
    when:
      any_changed:
        - "src/routes/**"
        - "openapi/**"
        - "proto/**"
        - "src/controllers/**"
        - "app/controllers/**"
    require:
      groups:
        - name: "API 계약 문서"
          any_changed:
            - "docs/spec.md"
            - "docs/api/**"
        - name: "릴리즈 공지"
          all_changed:
            - "CHANGELOG.md"
    severity: blocker
    message: "API surface changed without synced contract/docs"

  # ─── DB 스키마 변경 → 운영 문서 + 검증 흔적 ──────────────────────────────────
